MinimizeBtn.MouseButton1Click:Connect(function()
	isMinimized = not isMinimized
	if isMinimized then
		ContentFrame.Visible = false
		MainFrame.Size = UDim2.new(0, 280, 0, 32)

	else
		ContentFrame.Visible = true
		MainFrame.Size = originalSize

	end
end)

-- Spawn Platform
local function spawnPlatform()
	if spawnedPart then
		spawnedPart:Destroy()
	end

	local hrp = HumanoidRootPart
	if not hrp then return end

	local part = Instance.new("Part")
	part.Name = "SpawnedPlatform"
	part.Size = Vector3.new(30, currentThickness, 30) -- wide + thick
	part.Position = hrp.Position - Vector3.new(0, 30 + (currentThickness / 2), 0)
	part.Anchored = true
	part.CanCollide = true
	part.Material = Enum.Material.SmoothPlastic
	part.Color = Color3.fromRGB(80, 80, 90)
	part.Parent = workspace

	spawnedPart = part
	StatusLabel.Text = "Platform spawned"
	StatusLabel.TextColor3 = Color3.fromRGB(0, 200, 100)
end

-- Update thickness
local function updateThickness(amount)
	currentThickness = math.clamp(currentThickness + amount, 1, 50)
	ThicknessLabel.Text = "Thickness: " .. currentThickness

	if spawnedPart and spawnedPart.Parent then
		local oldSize = spawnedPart.Size
		local oldPos = spawnedPart.Position
		spawnedPart.Size = Vector3.new(30, currentThickness, 30)
		-- Keep the top of the part in the same place when resizing
		spawnedPart.Position = Vector3.new(oldPos.X, oldPos.Y + (currentThickness - oldSize.Y) / 2, oldPos.Z)
	end
end

-- Make part above see-through + walkable through
local function makeAboveSeeThrough()
	local hrp = HumanoidRootPart
	if not hrp then return end

	local origin = hrp.Position
	local direction = Vector3.new(0, 50, 0) -- look upward

	local rayParams = RaycastParams.new()
	rayParams.FilterDescendantsInstances = {Character}
	rayParams.FilterType = Enum.RaycastFilterType.Exclude

	local result = workspace:Raycast(origin, direction, rayParams)

	if result and result.Instance and result.Instance:IsA("BasePart") then
		local part = result.Instance
		part.Transparency = 0.5
		part.CanCollide = false
		StatusLabel.Text = "Made " .. part.Name .. " see-through"
		StatusLabel.TextColor3 = Color3.fromRGB(0, 200, 100)
	else
		StatusLabel.Text = "No part found above you"
		StatusLabel.TextColor3 = Color3.fromRGB(255, 80, 80)
	end
end
