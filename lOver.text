local function changeFloorThickness(amount)
	if not floorPart or not floorPart.Parent then
		floorThickness = math.clamp(floorThickness + amount, 1, 80)
		FloorThickLabel.Text = "Floor Thickness: " .. floorThickness
		return
	end

	-- Get the ACTUAL current floor size
	local currentSize = floorPart.Size

	-- Change ONLY Y thickness
	local newThickness = math.clamp(
		currentSize.Y + amount,
		1,
		80
	)

	-- Remember the bottom of the floor
	local bottomY =
		floorPart.Position.Y -
		(currentSize.Y / 2)

	-- Keep X and Z exactly the same
	floorPart.Size = Vector3.new(
		currentSize.X,
		newThickness,
		currentSize.Z
	)

	-- Move it so the bottom stays in the same place
	floorPart.Position = Vector3.new(
		floorPart.Position.X,
		bottomY + (newThickness / 2),
		floorPart.Position.Z
	)

	-- Update stored value
	floorThickness = newThickness

	-- Update GUI
	FloorThickLabel.Text =
		"Floor Thickness: " .. newThickness

	StatusLabel.Text =
		"Floor thickness: " .. newThickness

	StatusLabel.TextColor3 =
		Color3.fromRGB(0, 200, 100)
end
