-- ========== FUNCTIONS ==========

-- Spawn super wide floor
local function spawnFloor()
	if floorPart then floorPart:Destroy() end

	local hrp = HumanoidRootPart
	if not hrp then return end

	local part = Instance.new("Part")
	part.Name = "SuperFloor"
	part.Size = Vector3.new(200, floorThickness, 200) -- SUPER WIDE
	part.Position = hrp.Position - Vector3.new(0, 30 + (floorThickness / 2), 0)
	part.Anchored = true
	part.CanCollide = true
	part.Material = Enum.Material.SmoothPlastic
	part.Color = Color3.fromRGB(70, 70, 80)
	part.Parent = workspace

	floorPart = part
	StatusLabel.Text = "Floor spawned (200x200)"
	StatusLabel.TextColor3 = Color3.fromRGB(0, 200, 100)
end

-- Spawn sky crush block
local function spawnSky()
	if skyPart then skyPart:Destroy() end

	local hrp = HumanoidRootPart
	if not hrp then return end

	local part = Instance.new("Part")
	part.Name = "SkyCrush"
	part.Size = Vector3.new(200, skyThickness, 200) -- SUPER WIDE
	part.Position = hrp.Position + Vector3.new(0, 25 + (skyThickness / 2), 0)
	part.Anchored = true
	part.CanCollide = true
	part.Material = Enum.Material.SmoothPlastic
	part.Color = Color3.fromRGB(120, 40, 160)
	part.Parent = workspace

	skyPart = part
	StatusLabel.Text = "Sky crush block spawned"
	StatusLabel.TextColor3 = Color3.fromRGB(200, 100, 255)
end

-- Floor thickness (grows both ways / keeps center)
local function changeFloorThickness(amount)
	floorThickness = math.clamp(floorThickness + amount, 1, 80)
	FloorThickLabel.Text = "Floor Thickness: " .. floorThickness

	if floorPart and floorPart.Parent then
		local oldSize = floorPart.Size
		floorPart.Size = Vector3.new(200, floorThickness, 200)
		-- Keep top surface in place
		floorPart.Position = floorPart.Position + Vector3.new(0, (floorThickness - oldSize.Y) / 2, 0)
	end
end

-- Sky thickness (grows DOWNWARD only = crush effect)
local function changeSkyThickness(amount)
	skyThickness = math.clamp(skyThickness + amount, 1, 100)
	SkyThickLabel.Text = "Sky Thickness: " .. skyThickness

	if skyPart and skyPart.Parent then
		local oldSize = skyPart.Size
		local topY = skyPart.Position.Y + (oldSize.Y / 2) -- keep the top fixed
		skyPart.Size = Vector3.new(200, skyThickness, 200)
		skyPart.Position = Vector3.new(skyPart.Position.X, topY - (skyThickness / 2), skyPart.Position.Z)
	end
end

local function deleteSky()
	if skyPart then
		skyPart:Destroy()
		skyPart = nil
		StatusLabel.Text = "Sky block deleted"
		StatusLabel.TextColor3 = Color3.fromRGB(255, 100, 100)
	end
end

-- Buttons
SpawnFloorBtn.MouseButton1Click:Connect(spawnFloor)
FloorPlus.MouseButton1Click:Connect(function() changeFloorThickness(2) end)
FloorMinus.MouseButton1Click:Connect(function() changeFloorThickness(-2) end)

SpawnSkyBtn.MouseButton1Click:Connect(spawnSky)
SkyPlus.MouseButton1Click:Connect(function() changeSkyThickness(3) end) -- bigger steps for crush
SkyMinus.MouseButton1Click:Connect(function() changeSkyThickness(-3) end)
DeleteSkyBtn.MouseButton1Click:Connect(deleteSky)
